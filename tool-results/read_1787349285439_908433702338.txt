   860→    function trapFocus(modal) {
   861→        const focusableElements = modal.querySelectorAll(
   862→            'button, [href], input, select, textarea, [tabindex]:not([tabindex="-1"])'
   863→        );
   864→        
   865→        if (focusableElements.length === 0) return;
   866→        
   867→        const firstElement = focusableElements[0];
   868→        const lastElement = focusableElements[focusableElements.length - 1];
   869→
   870→        friendModalState.keydownHandler = function (e) {
   871→            if (e.key === 'Tab') {
   872→                if (e.shiftKey) {
   873→                    if (document.activeElement === firstElement) {
   874→                        lastElement.focus();
   875→                        e.preventDefault();
   876→                    }
   877→                } else {
   878→                    if (document.activeElement === lastElement) {
   879→                        firstElement.focus();
   880→                        e.preventDefault();
   881→                    }
   882→                }
   883→            }
   884→            if (e.key === 'Escape') {
   885→                hideFriendModal();
   886→            }
   887→        };
   888→
   889→        modal.addEventListener('keydown', friendModalState.keydownHandler);
   890→        firstElement.focus();
   891→    }
